import { EvEm } from '@jcfigueiredo/evem';
import { SseHandler } from '@jcfigueiredo/evem/sse';

const evem = new EvEm();

// Lifecycle and errors are EvEm events too (subscribe before connecting to see "connecting")
evem.subscribe('sse.connection.state', ({ from, to }) => showStatus(from, to));
evem.subscribe('sse.error', ({ reason, status, error }) => console.warn('SSE', reason, status, error.message));
evem.subscribe('sse.parse.error', ({ eventType, rawData }) => console.warn('Not JSON:', eventType, rawData));
evem.subscribe('sse.reconnect.failed', ({ attempts }) => showOffline(attempts));

// Connects right away. headers() runs before every (re)connect, so a refreshed token is
// always sent; when reconnecting, the Last-Event-ID header is added
const sse = new SseHandler('/api/events', evem, {
  headers: () => ({ Authorization: `Bearer ${getToken()}` }),
  heartbeatTimeout: 4000,    // nothing (not even ": ping") for 4 s: reconnect
  reconnectDelay: 1000,      // base delay; the server's "retry:" replaces it
  maxReconnectDelay: 8000,   // the delay doubles per failed attempt (±20% jitter), up to this
  maxReconnectAttempts: 4,   // then sse.reconnect.failed, and stop
});

// To resume after a reload, save the id and pass it back: new SseHandler(url, evem, { lastEventId })
window.addEventListener('pagehide', () => {
  const lastEventId = sse.getLastEventId();
  if (lastEventId) sessionStorage.setItem('orders-stream:last-event-id', lastEventId);
});

async function signOut() {
  await sse.disconnect();   // closes the stream and cancels a pending reconnect
}

This page creates its handler with exactly these options, plus fetch: server.fetch (the simulated server instead of the network) and rawEvents: true (to list the event ids above). To retry a 401 with a fresh token instead of stopping, pass a shouldReconnect(info) that handles that case and returns defaultShouldReconnect(info) (also exported by @jcfigueiredo/evem/sse) for the rest. Outside a browser, give the handler an absolute URL.

Simulated server

Send

A named event and an envelope both become server.order.*; an unnamed message without an envelope becomes sse.message; data that isn't JSON becomes sse.parse.error.

Stream

With heartbeats off, the connection is silent between price ticks; after 4 s without a byte the client gives up on it and reconnects. Small chunks cut lines (and UTF-8 characters) across reads, marked ┊ on the wire.

Failures


"Restart" closes the stream; the client reconnects and gets that answer. 401 (and other 4xx) stop the client with sse.error; 503 makes it back off (at least Retry-After); 204 stops it quietly. Unreachable for long enough, it gives up with sse.reconnect.failed.

import { createServer } from 'node:http';
import { formatSseComment, formatSseMessage, SSE_HEADERS } from '@jcfigueiredo/evem/sse/server';

const clients = new Set();
const history = [];   // recent events, to replay what a reconnecting client missed
let nextId = 1;

function broadcast(event, data) {
  const message = { id: nextId++, event, data };
  history.push(message);
  if (history.length > 1000) history.shift();
  for (const res of clients) res.write(formatSseMessage(message));
}

createServer((req, res) => {
  if (req.url !== '/api/events') {
    res.writeHead(404).end();
    return;
  }
  if (!isAuthorized(req.headers.authorization)) {
    res.writeHead(401).end();   // the client stops: retrying won't help
    return;
  }

  res.writeHead(200, SSE_HEADERS);
  res.write(formatSseMessage({ retry: 2000 }));   // the client's base reconnect delay

  const lastEventId = req.headers['last-event-id'];
  if (lastEventId) {
    // Resume: replay the events after the client's Last-Event-ID
    for (const message of history) {
      if (message.id > Number(lastEventId)) res.write(formatSseMessage(message));
    }
  } else {
    // A new client: an id without data isn't an event, but it tells the client where it starts
    res.write(formatSseMessage({ id: nextId - 1 }));
  }

  clients.add(res);
  const ping = setInterval(() => res.write(formatSseComment('ping')), 15_000);   // heartbeat
  req.on('close', () => {
    clearInterval(ping);
    clients.delete(res);
  });
}).listen(3000);

formatSseMessage JSON-encodes data, writes one data: line per line (so content can't forge events) and ends with the blank line that dispatches the event.

# evem_sse.py is the reference helper in examples/python/: copy it next to this file
import asyncio
import re

from fastapi import FastAPI, Request
from fastapi.responses import Response, StreamingResponse

from evem_sse import SSE_HEADERS, format_sse_comment, format_sse_message

app = FastAPI()
subscribers = set()   # one asyncio.Queue per open stream
history = []          # recent events, to replay what a reconnecting client missed
next_id = 1


def broadcast(event, data):
    global next_id
    message = {"id": next_id, "event": event, "data": data}
    next_id += 1
    history.append(message)
    del history[:-1000]
    for queue in subscribers:
        queue.put_nowait(message)


@app.get("/api/events")
async def events(request: Request):
    if not is_authorized(request.headers.get("authorization")):
        return Response(status_code=401)   # the client stops: retrying won't help
    last_event_id = request.headers.get("last-event-id", "")
    resume_after = int(last_event_id) if re.fullmatch(r"[0-9]+", last_event_id) else None

    async def stream():
        queue = asyncio.Queue()
        subscribers.add(queue)
        # Taken together with subscribing, so no event is both replayed and queued
        missed = [m for m in history if resume_after is not None and m["id"] > resume_after]
        start = next_id - 1
        try:
            yield format_sse_message(retry=2000)   # the client's base reconnect delay
            if resume_after is None:
                yield format_sse_message(id=start)  # a new client: where it starts (no data, no event)
            for message in missed:                  # resume: replay what the client missed
                yield format_sse_message(**message)
            while not await request.is_disconnected():
                try:
                    message = await asyncio.wait_for(queue.get(), timeout=15)
                except asyncio.TimeoutError:
                    yield format_sse_comment("ping")   # heartbeat
                else:
                    yield format_sse_message(**message)
        finally:
            subscribers.discard(queue)

    return StreamingResponse(stream(), headers=SSE_HEADERS)
